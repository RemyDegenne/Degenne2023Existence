\chapter{Fixed-budget complexity}
\label{chap:prereq_complexity}

Library-shaped material on fixed-budget algorithm families and static proportions, for
\texttt{ForLMLExplore/Identification/}. The definitions are those of Chapter~\ref{chap:setting}.

\begin{lemma}
  \label{lem:inv_difficultyRatio}
  \lean{LMLExplore.Identification.IdentTask.inv_difficultyRatio}
  \leanok
  \uses{def:difficulty_ratio}
  If $H(\theta) \ne 0, \infty$ and $T > 0$, then $R_{H,T}(\cA, \theta)^{-1} = H(\theta)
  \log(1/p_{\theta,T}(\cA)) / T$.
\end{lemma}

\begin{proof}
  \leanok
  The definitions, in $[0, \infty]$.
\end{proof}

\begin{lemma}
  \label{lem:tendsto_errorRate_iff}
  \uses{def:error_rate}
  For $c \in [0, \infty]$, $h_{\theta, T}(\cA) \to c^{-1}$ if and only if
  $-\frac1T \log p_{\theta, T}(\cA) \to c$.
\end{lemma}

\begin{proof}
  $h_{\theta, T} = (-\frac1T \log p_{\theta, T})^{-1}$ for $T > 0$, and inversion is a
  homeomorphism of $[0, \infty]$.
\end{proof}

\begin{lemma}
  \label{lem:oracleDifficulty_mono}
  \uses{def:oracle_difficulty}
  If $\cC \subseteq \cC'$, then $H_{\cC'} \le H_{\cC}$, and $R_{H_{\cC}, T} \le R_{H_{\cC'}, T}$.
\end{lemma}

\begin{proof}
  An infimum over a larger set is smaller.
\end{proof}

\begin{lemma}
  \label{lem:iSup_open_simplex}
  \uses{def:static_difficulty}
  The supremum of $\omega \mapsto \inf_{\lambda \in \Alt(\theta)} \sum_k \omega_k
  \KL(\nu_k(\lambda), \nu_k(\theta))$ over the positive proportions $\Simplex^0$ is its supremum
  over $\Simplex$.
\end{lemma}

\begin{proof}
  The function is concave and nonnegative: along the segment from $\omega \in \Simplex$ to the
  uniform allocation, its values are at least $(1 - t)$ times its value at $\omega$.
\end{proof}

\begin{lemma}
  \label{lem:iSup_iInf_mul}
  \lean{LMLExplore.iSup_iInf_mul_ofReal_le}
  \leanok
  For a finite set $s$ of arms and $c_j \in [0, \infty]$,
  $\sup_{\omega \in \Simplex} \min_{j \in s} c_j \omega_j \le (\sum_{j \in s} c_j^{-1})^{-1}$.
\end{lemma}

The bound is an equality when the $c_j$ are positive and finite: equalize $c_j \omega_j$ on $s$
and put no weight outside $s$. Corollary~\ref{cor:corner_lb} uses only the inequality.

\begin{proof}
  \leanok
  For $\omega \in \Simplex$ and $m = \min_{j \in s} c_j \omega_j$,
  $m \sum_{j \in s} c_j^{-1} \le \sum_{j \in s} \omega_j c_j c_j^{-1} \le \sum_{j \in s} \omega_j
  \le 1$, with $c c^{-1} \le 1$ in $[0, \infty]$.
\end{proof}

\begin{lemma}
  \label{lem:staticDifficulty_eq_charTime}
  \uses{def:static_difficulty}
  If $\KL(\nu_k(\lambda), \nu_k(\theta)) = \KL(\nu_k(\theta), \nu_k(\lambda))$ for all models and
  arms (Gaussian arms with known variances), then for a single-answer task $\Hsp(\theta)$ is the
  characteristic time $T^{\star}(\theta)$ of LMLExplore at every admissible model.
\end{lemma}

\begin{proof}
  Both are the inverse of the same maximin, the alternatives to $\theta$ being the alternatives to
  its answer.
\end{proof}

\begin{lemma}
  \label{lem:exists_static}
  \uses{def:static_proportions}
  For every $\omega \in \Simplex^0$ and every measurable recommendation rule, there is a static
  proportions family with proportions $\omega$ and that rule.
\end{lemma}

\begin{proof}
  Cumulative tracking with the constant targets $\omega$ (LMLExplore,
  \texttt{Identification/CumulativeTracking.lean}) has $\abs{N_{T,k} - T \omega_k} \le K - 1$
  almost surely; the fixed-budget algorithm with budget $T$ and the deterministic output rule
  $\mathrm{rec}(\hat\mu_T)$ (\texttt{fixedBudget}) completes it.
\end{proof}

\begin{lemma}
  \label{lem:fb_change_of_measure}
  \lean{LMLExplore.Identification.IsFixedBudget.klDiv_outputMeasure_le}
  \leanok
  \uses{def:fixed_budget_family}
  For an algorithm $\cA_T$ with budget $T$ and two models $\mu, \lambda$, the relative entropy
  between the laws of the output of $\cA_T$ on $\mu$ and on $\lambda$ is at most
  $\sum_k \E_{\mu}[N_{T,k}] \KL(\nu_k(\mu), \nu_k(\lambda))$. Hence for every measurable event $E$
  of the output, $\kl(\Prob_{\mu}(E), \Prob_{\lambda}(E)) \le \sum_k \E_{\mu}[N_{T,k}]
  \KL(\nu_k(\mu), \nu_k(\lambda))$.
\end{lemma}

\begin{proof}
  \leanok
  The divergence decomposition at the stopping time, which is $T$ (LMLExplore's
  \texttt{Identification/ChangeOfMeasure.lean}), and the data processing inequality
  (LMLExplore's \texttt{klBer\_measureReal\_le\_klDiv}).
\end{proof}

\begin{lemma}
  \label{lem:klBer_ge}
  \lean{LMLExplore.mul_neg_log_le_klBerReal_add_log_two,
    LMLExplore.ofReal_mul_neg_log_le_klBer_add_log_two}
  \leanok
  For $a, b \in [0, 1]$, $\kl(a, b) \ge a \log(1/b) - \log 2$, with $\log(1/0) = \infty$.
\end{lemma}

\begin{proof}
  \leanok
  $\kl(a, b) = a \log\frac1b + (1 - a) \log\frac1{1 - b} - H(a) \ge a \log\frac1b - \log 2$, with
  $H(a) \le \log 2$ the binary entropy. For $b = 0 < a$ both sides are infinite.
\end{proof}

\begin{lemma}
  \label{lem:tendsto_subseq_proportions}
  \lean{LMLExplore.Identification.IsFixedBudget.expectedPullProportion_mem_simplex,
    LMLExplore.Identification.IsFixedBudget.lintegral_pullCount_div}
  \leanok
  For an algorithm with budget $T \ge 1$, the expected proportions $\E_{\mu}[N_{T,k}] / T$ are
  LMLExplore's expected proportions of pulls, and lie in $\Simplex$. Hence for a fixed-budget
  family, every sequence of budgets has a subsequence along which they converge in $\Simplex$.
\end{lemma}

\begin{proof}
  \leanok
  The expected stopping time is $T$; the simplex is compact (LMLExplore's
  \texttt{isCompact\_simplex}).
\end{proof}

\section{Gaussian arms and half-spaces}

For \texttt{ForLMLExplore/Identification/Task/}: Gaussian arms with known variances $\sigma_k^2 =
v_k > 0$, and half-space identification with normal vector $u \ne 0$. Write $L = \sum_k \abs{u_k}
\sigma_k$.

\begin{lemma}
  \label{lem:cost_gaussianReal}
  \lean{LMLExplore.Identification.IdentTask.cost_gaussianReal,
    LMLExplore.Identification.IdentTask.cost_comm_gaussianReal}
  \leanok
  For Gaussian arms with variances $v_k > 0$ and $\omega \ge 0$,
  $\sum_k \omega_k \KL(\nu_k(\mu), \nu_k(\lambda)) = \sum_k \omega_k (\mu_k - \lambda_k)^2 / (2
  v_k)$, which is symmetric in $\mu$ and $\lambda$.
\end{lemma}

\begin{proof}
  \leanok
  $\KL(\mathcal N(x, v), \mathcal N(y, v)) = (x - y)^2 / (2 v)$ (LMLExplore).
\end{proof}

\begin{lemma}
  \label{lem:alternatives_halfSpace}
  \lean{LMLExplore.Identification.mem_alternatives_halfSpaceTask_iff}
  \leanok
  \uses{def:alternatives}
  For half-space identification and $u^{\top} \mu \ne 0$, $\lambda \in \Alt(\mu)$ if and only if
  $(u^{\top} \mu)(u^{\top} \lambda) < 0$.
\end{lemma}

\begin{proof}
  \leanok
  The correct answer is the sign of $u^{\top} \mu$.
\end{proof}

\begin{lemma}
  \label{lem:halfspace_direction}
  \lean{LMLExplore.Identification.exists_dotProduct_eq_one_sum_le}
  \leanok
  The direction $d_k = \mathrm{sign}(u_k) \sigma_k / L$ has $u^{\top} d = 1$, and for every $\omega
  \in \Simplex$ and $s \in \R$, $\sum_k \omega_k (s d_k)^2 / (2 v_k) \le s^2 / (2 L^2)$.
\end{lemma}

\begin{proof}
  \leanok
  $u_k d_k = \abs{u_k} \sigma_k / L$, and $(s d_k)^2 / v_k \le s^2 / L^2$.
\end{proof}

\begin{lemma}
  \label{lem:halfspace_cauchy_schwarz}
  \lean{LMLExplore.Identification.sq_dotProduct_div_le_sum}
  \leanok
  If $(u^{\top} x)(u^{\top} y) \le 0$, then at $\omega^{\star}_k = \abs{u_k} \sigma_k / L$,
  $(u^{\top} x)^2 / (2 L^2) \le \sum_k \omega^{\star}_k (y_k - x_k)^2 / (2 v_k)$.
\end{lemma}

\begin{proof}
  \leanok
  $(u^{\top} x)^2 \le (u^{\top}(x - y))^2 \le (\sum_k \abs{u_k} \sigma_k L) (\sum_k \abs{u_k}
  \sigma_k (x_k - y_k)^2 / (L v_k))$ by the Cauchy--Schwarz inequality.
\end{proof}

\begin{lemma}
  \label{lem:halfspace_static}
  \lean{LMLExplore.Identification.inv_staticDifficulty_halfSpaceTask_gaussianReal,
    LMLExplore.Identification.staticDifficulty_halfSpaceTask_gaussianReal,
    LMLExplore.Identification.exists_mem_alternatives_halfSpaceTask_gaussianReal,
    LMLExplore.Identification.ofReal_le_cost_halfSpaceTask_gaussianReal}
  \leanok
  \uses{def:static_difficulty, lem:cost_gaussianReal, lem:alternatives_halfSpace,
    lem:halfspace_direction, lem:halfspace_cauchy_schwarz}
  For Gaussian half-space identification and $u^{\top} \mu \ne 0$,
  $\Hsp(\mu)^{-1} = (u^{\top} \mu)^2 / (2 L^2)$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:cost_gaussianReal, lem:alternatives_halfSpace, lem:halfspace_direction,
    lem:halfspace_cauchy_schwarz}
  Upper bound: for $t > 1$, the alternative $\mu - t (u^{\top} \mu) d$
  (Lemma~\ref{lem:halfspace_direction}) costs at most $t^2 (u^{\top} \mu)^2 / (2 L^2)$ for every
  allocation; let $t \downarrow 1$. Lower bound: at $\omega^{\star}$, every alternative costs at
  least $(u^{\top} \mu)^2 / (2 L^2)$ (Lemma~\ref{lem:halfspace_cauchy_schwarz}).
\end{proof}

\begin{lemma}
  \label{lem:halfspace_ratio}
  \lean{LMLExplore.Identification.iSup_iInf_staticDifficulty_mul_cost_halfSpaceTask_le_one}
  \leanok
  \uses{lem:halfspace_static, lem:halfspace_direction}
  For Gaussian half-space identification and $u^{\top} \mu \ne 0$,
  $\max_{\omega \in \Simplex} \inf_{\lambda \in \Alt(\mu)} \Hsp(\lambda) \sum_k \omega_k
  \KL(\nu_k(\mu), \nu_k(\lambda)) \le 1$.
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:halfspace_static, lem:halfspace_direction}
  At $\lambda = \mu - t (u^{\top} \mu) d$, $\Hsp(\lambda) = 2 L^2 / ((t - 1)^2 (u^{\top} \mu)^2)$
  (Lemma~\ref{lem:halfspace_static}) and the cost is at most $t^2 (u^{\top} \mu)^2 / (2 L^2)$, so
  the product is at most $t^2 / (t - 1)^2$, which tends to $1$ as $t \to \infty$.
\end{proof}
